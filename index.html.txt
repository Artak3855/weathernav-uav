    <script>
        function calculateAll() {
            // Կարդում ենք մուտքային արժեքները
            const f = parseFloat(document.getElementById('freq').value); // ԳՀց
            const d1 = parseFloat(document.getElementById('d1').value); // կմ
            const d2 = parseFloat(document.getElementById('d2').value); // կմ
            const h_obs = parseFloat(document.getElementById('h_obs').value); // մետր
            
            const ptx = parseFloat(document.getElementById('ptx').value);
            const gtx = parseFloat(document.getElementById('gtx').value);
            const grx = parseFloat(document.getElementById('grx').value);
            const losses = parseFloat(document.getElementById('losses').value);
            const sensitivity = parseFloat(document.getElementById('sensitivity').value);

            const totalD = d1 + d2; // Ընդհանուր հեռավորություն կմ

            // 1. Ֆրենելի առաջին գոտու շառավիղը խոչընդոտի կետում (մետրերով)
            // R1 = 17.32 * sqrt( (d1 * d2) / (f * (d1 + d2)) )
            let r1 = 17.32 * Math.sqrt((d1 * d2) / (f * totalD));
            let threshold60 = 0.6 * r1;

            document.getElementById('res_r1').innerText = r1.toFixed(2) + ' մ';
            document.getElementById('res_60').innerText = threshold60.toFixed(2) + ' մ';

            let obsStatusElem = document.getElementById('res_obs_status');
            if (h_obs <= threshold60) {
                obsStatusElem.innerText = 'Անվտանգ (Կապն աշխատելու է)';
                obsStatusElem.className = 'text-lg font-extrabold text-green-400';
            } else if (h_obs <= r1) {
                obsStatusElem.innerText = 'Զգուշացում (Մտել է 60%-ի մեջ)';
                obsStatusElem.className = 'text-lg font-extrabold text-amber-400';
            } else {
                obsStatusElem.innerText = 'Վտանգավոր է (Խոչընդոտում է կապին)';
                obsStatusElem.className = 'text-lg font-extrabold text-red-400';
            }

            // 2. Link Budget Հաշվարկ
            // FSPL (dB) = 20*log10(d_km) + 20*log10(f_GHz) + 92.45
            let fspl = 20 * Math.log10(totalD) + 20 * Math.log10(f) + 92.45;
            
            // Prx = Ptx + Gtx + Grx - FSPL - losses
            let prx = ptx + gtx + grx - fspl - losses;

            // Link Margin = Prx - Sensitivity
            let linkMargin = prx - sensitivity;

            document.getElementById('res_fspl').innerText = fspl.toFixed(2) + ' dB';
            document.getElementById('res_prx').innerText = prx.toFixed(2) + ' dBm';
            
            let marginElem = document.getElementById('res_margin');
            marginElem.innerText = linkMargin.toFixed(2) + ' dB';
            if (linkMargin >= 10) {
                marginElem.className = 'text-2xl font-extrabold text-green-400';
            } else if (linkMargin >= 0) {
                marginElem.className = 'text-2xl font-extrabold text-amber-400';
            } else {
                marginElem.className = 'text-2xl font-extrabold text-red-400';
            }

            // Վիզուալիզացիա Canvas-ի վրա
            drawFresnel(d1, d2, r1, h_obs);
        }

        function drawFresnel(d1, d2, r1, h_obs) {
            const canvas = document.getElementById('fresnelCanvas');
            const ctx = canvas.getContext('2d');
            const width = canvas.width;
            const height = canvas.height;

            ctx.clearRect(0, 0, width, height);

            let startX = 50;
            let endX = width - 50;
            let centerY = height / 2;
            let maxRadiusPx = 70; // Էլիպսի նկարվող առավելագույն բարձրությունը պիքսելներով

            // Նկարում ենք Ուղղակի տեսանելիության գիծը (LoS)
            ctx.beginPath();
            ctx.strokeStyle = '#94a3b8';
            ctx.setLineDash([4, 4]);
            ctx.moveTo(startX, centerY);
            ctx.lineTo(endX, centerY);
            ctx.stroke();
            ctx.setLineDash([]); // վերականգնել

            // Նկարում ենք Ֆրենելի էլիպսը
            ctx.beginPath();
            ctx.strokeStyle = '#06b6d4';
            ctx.lineWidth = 2;
            // Օգտագործում ենք ellipse մետոդը
            let controlX = startX + (endX - startX) * (d1 / (d1 + d2));
            ctx.ellipse((startX + endX) / 2, centerY, (endX - startX) / 2, maxRadiusPx, 0, 0, 2 * Math.PI);
            ctx.fillStyle = 'rgba(6, 182, 212, 0.05)';
            ctx.fill();
            ctx.stroke();

            // Նկարում ենք 60% անվտանգության գիծը ներսում
            ctx.beginPath();
            ctx.strokeStyle = 'rgba(245, 158, 11, 0.5)';
            ctx.ellipse((startX + endX) / 2, centerY, (endX - startX) / 2, maxRadiusPx * 0.6, 0, 0, 2 * Math.PI);
            ctx.stroke();

            // Խոչընդոտի դիրքը x-ի առանցքով
            let obsX = controlX;
            // Հաշվենք խոչընդոտի բարձրության պիքսելային չափը վիզուալիզացիայի համար
            let obsHeightPx = Math.min((h_obs / r1) * maxRadiusPx, maxRadiusPx + 15);

            // Նկարում ենք խոչընդոտը (սուր սյուն ներքևից կամ վերևից)
            ctx.beginPath();
            ctx.strokeStyle = h_obs <= (0.6 * r1) ? '#22c55e' : (h_obs <= r1 ? '#f59e0b' : '#ef4444');
            ctx.lineWidth = 4;
            ctx.moveTo(obsX, centerY);
            ctx.lineTo(obsX, centerY - obsHeightPx);
            ctx.stroke();

            // Աշտարակներ (Tx / Rx)
            ctx.fillStyle = '#f8fafc';
            ctx.fillRect(startX - 4, centerY - 25, 8, 25);
            ctx.fillRect(endX - 4, centerY - 25, 8, 25);

            ctx.fillStyle = '#94a3b8';
            ctx.font = '12px sans-serif';
            ctx.fillText('Tx', startX - 8, centerY + 15);
            ctx.fillText('Rx', endX - 8, centerY + 15);
        }

        // Առաջնային հաշվարկ բացվելիս
        window.onload = calculateAll;
    </script>
